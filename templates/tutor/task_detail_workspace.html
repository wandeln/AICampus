  // Speicher: DB/LLM liefern „4g“-Strings → UI zeigt eine GB-Zahl
  function wsMemToGb(v) {
    const s = String(v == null ? "" : v).trim().toLowerCase();
    const n = parseFloat(s);
    if (!isFinite(n)) return "";
    let gb = n;
    if (s.endsWith("m")) gb = n / 1000;
    else if (s.endsWith("k")) gb = n / 1e6;
    else if (s.endsWith("b")) gb = n / 1e9;
    return String(Math.round(gb * 10) / 10);
  }

  // Engine-Optionen rendern (gefiltert auf Engines mit installiertem Image)
  function wsRenderEngineOptions() {
    const current = wsEngineSel.value;
    wsEngineSel.innerHTML = '';
    let candidates = WS_ENGINES;
    if (WS_INSTALLED) {
      candidates = WS_ENGINES.filter(a => WS_INSTALLED.has(a.name));
      if (!candidates.length) {
        // Nur die 📭-Option anzeigen (disabled → value bleibt "" → Save-Hook liefert [])
        wsEngineSel.innerHTML = "";
        const opt = document.createElement("option");
        opt.value = "";
        opt.disabled = true;
        opt.selected = true;
        opt.textContent = "📭 Image wurde noch auf keiner Engine installiert";
        wsEngineSel.appendChild(opt);
        wsEngineHintEl.textContent =
          "Bitte das Image zuerst auf einer Engine installieren (Kurs-Einstellungen oder Admin-Konsole → Compute/Workspace).";
        return;
      }
    }
    candidates.forEach(a => {
      const opt = document.createElement("option");
      opt.value = a.name;
      opt.textContent = wsEngineLabel(a);
      wsEngineSel.appendChild(opt);
    });
    // Gespeicherte Auswahl beibehalten, falls noch verfügbar
    const initEng = (WS_ENGINES_INIT && WS_ENGINES_INIT.length) ? String(WS_ENGINES_INIT[0]) : "";
    const wanted = candidates.some(a => a.name === current) ? current
                 : (candidates.some(a => a.name === initEng) ? initEng : "");
    if (wanted) wsEngineSel.value = wanted;
    wsEngineHintEl.textContent = WS_INSTALLED
      ? "Nur Engines mit installiertem Image der gewählten Spec · GPU wird automatisch erkannt."

  async function wsLoadEngines() {
    try {
      const res = await fetch("/api/courses/" + courseId + "/compute/engines", { credentials: "same-origin" });
      if (!res.ok) { return; }
      const data = await res.json();
      WS_ENGINES = data.agents || [];
      wsRenderEngineOptions();
    } catch (e) { /* Engine-Liste bleibt leer (degradiert) */ }
  }

  // Prüft, auf welchen Engines die gewählte Spec installiert ist
  // (spec_status; Fallback: images[].spec_name).
  // Offline Engines können nicht verifiziert werden → fallen im Filter-Fall weg.
  function wsRefreshInstalled() {
    wsEngineRefreshP = (async () => {
      const spec = WS_SPECS.find(x => x.name === wsSpecSel.value);
      if (!spec) { WS_INSTALLED = null; wsRenderEngineOptions(); return; }
      const results = await Promise.all(WS_ENGINES.map(async (a) => {
        try {
          const res = await fetch(
            "/api/courses/" + courseId + "/compute/engines/" + encodeURIComponent(a.name) + "/images",
            { credentials: "same-origin" });
          if (!res.ok) throw new Error(res.status);
          const data = await res.json();
          const st = (data.spec_status || {})[String(spec.id)];
          if (st) return { name: a.name, installed: !!st.installed, ok: true };
          // Fallback: spec_status leer (z. B. Agent ohne Support) → Image-Liste vergleichen
          return { name: a.name, installed: (data.images || []).some(i => i.spec_name === spec.name), ok: true };
        } catch (e) {
          return { name: a.name, installed: false, ok: false }; // offline/unverfügbar
        }
      }));
      WS_INSTALLED = new Set(results.filter(r => r.ok && r.installed).map(r => r.name));
      wsRenderEngineOptions();
    })();
    return wsEngineRefreshP;
  }

  async function wsLoadSpecs(selectName) {
    try {
      const res = await fetch("/api/courses/" + courseId + "/image-specs", { credentials: "same-origin" });
      if (!res.ok) { return; }
      const data = await res.json();
      WS_SPECS = data.specs || [];
      const current = selectName != null ? selectName : wsSpecSel.value;
      wsSpecSel.innerHTML = '';
      WS_SPECS.forEach(s => {
        const opt = document.createElement("option");
        opt.value = s.name;
        const scope = s.scope !== "global" ? "" : " (global)";

        wsSpecSel.value = current;
      }
      wsUpdateSpecInfo();
      wsRefreshInstalled();
    } catch (e) { /* Spec-Liste bleibt leer (degradiert) */ }
  }

  // ─── Umgebung-Status je Engine (Assets/Task-Image/IMAGE) ──────
  // Format: {agent_url: {assets, task_image, image, error?}}
  function wsChip(label, val) {
    let icon = "❓", cls = "bg-gray-50 border-gray-200 text-gray-500";
    if (val === "ready") { icon = "✅"; cls = "bg-green-50 border-green-200 text-green-700"; }
    else if (val === "pending") { icon = "⏳"; cls = "bg-amber-50 border-amber-200 text-amber-700"; }
    else if (val === "building") { icon = "🏗"; cls = "bg-amber-50 border-amber-200 text-amber-700"; }
    else if (val === "failed") { icon = "❌"; cls = "bg-red-50 border-red-200 text-red-700"; }
    else if (val === "none" || val === "skipped") { icon = "·"; cls = "bg-gray-50 border-gray-200 text-gray-400"; }
    return '' + icon + " " + label + "";
  }

  // Rendert die Status-Chips; liefert true, solange noch „pending" ist.
  function wsRenderAssetStatus(jsonStr) {
    const list = document.getElementById("ws-asset-status-list");
    let status = null;
    if (jsonStr) {
      try { status = JSON.parse(jsonStr); } catch (e) { status = null; }
    }
    if (!status || Object.keys(status).length === 0) {
      list.innerHTML = '(noch kein Status — nach dem Speichern mit „🔄 Sync & Init" auffüllen)';
      return false;
    }
    list.innerHTML = "";
    let pending = false;
    Object.entries(status).forEach(([url, st]) => {
      const name = url.replace(/^https?:\/\//, "").replace(/\/$/, "");
      const stObj = (typeof st === "object" && st) ? st
                   : { assets: st, task_image: null, image: null };
      const wrap = document.createElement("div");
      wrap.className = "flex items-center gap-1.5 flex-wrap border border-gray-200 rounded-lg px-2 py-1 bg-white";
      wrap.innerHTML = '' + escapeHtml(name) + "" +
        wsChip("Assets", stObj.assets) +
        wsChip("Task-Image", stObj.task_image) +
        wsChip("Image", stObj.image);
      if (stObj.error) wrap.title = stObj.error;
      if (JSON.stringify(stObj).includes("pending")) pending = true;
      list.appendChild(wrap);
    });
    return pending;
  }

  // Init-Log bleibt nach dem Build sichtbar (fertig = grau) und wird erst
  // bei einem neuen Build zurückgesetzt — sonst "verschwindet" es sofort.
  const wsInitLog = { text: "", building: false };

  function wsRenderInitLog(state, log) {
    const wrap = document.getElementById("ws-init-log-wrap");
    const pre = document.getElementById("ws-init-log");
    if (state === "building") {
      if (!wsInitLog.building) wsInitLog.text = "";  // neuer Build → Log klar
      wsInitLog.building = true;
      if (log) wsInitLog.text = log;
      pre.style.color = "#fcd34d";
      pre.textContent = wsInitLog.text || "(noch keine Ausgabe)";
    } else {
      wsInitLog.building = false;
      if (state === "failed") {
        wsInitLog.text = "❌ " + (log || "Init fehlgeschlagen");
        pre.style.color = "#f87171";
      } else if (state === "ready" && log) {
        wsInitLog.text = log;
        pre.style.color = "#86efac";
      } else if (wsInitLog.text) {
        pre.style.color = "#9ca3af";  // fertig ohne Tail: letzten Text behalten
      }
    }
    if (wsInitLog.text) {
      wrap.classList.remove("hidden");
      pre.textContent = wsInitLog.text;
      pre.scrollTop = pre.scrollHeight;
    } else {
      wrap.classList.add("hidden");
    }
    const stopBtn = document.getElementById("ws-init-stop-btn");
    if (stopBtn) stopBtn.classList.toggle("hidden", state !== "building");
  }

  let wsPollTimer = null;
  let wsPollTries = 0;
  let wsPrevInit = null;  // letzter Init-Status (für den Baum-Refresh)
  async function wsPollTick() {
    if (!WS_TASK_ID) { wsStopPolling(); return; }
    wsPollTries++;
    let assetsPending = false;
    let building = false;
    try {
      const r = await fetch("/api/tasks/" + WS_TASK_ID, { credentials: "same-origin" });
      const task = await r.json();
      assetsPending = wsRenderAssetStatus(task.workspace_assets_status || "");
    } catch (e) { /* weiter pollen */ }
    let log = "";
    let initState = "none";
    try {
      const r = await fetch(WS_TUTOR_FILES_API + "/init-status", { credentials: "same-origin" });
      const data = await r.json();
      const b = (data.agents || []).find(a => a.status === "building");
      const f = (data.agents || []).find(a => a.status === "failed");
      const rdy = (data.agents || []).find(a => a.status === "ready");
      if (b) { initState = "building"; log = b.log_tail || ""; }
      else if (f) { initState = "failed"; log = f.error || ""; }
      else if (rdy) { initState = "ready"; log = rdy.log_tail || ""; }
      building = (initState === "building");
      wsRenderInitLog(initState, log);
      // Init-Build gerade beendet (building → ready/failed/none) ODER der
      // erste Check zeigt schon „ready" (schneller Build) → [init]-Einträge
      // können neu/geändert sein → Datei-Baum aktualisieren (sonst erst
      // nach einem Seite-Reload).
      if ((wsPrevInit === "building" &&
           (initState === "ready" || initState === "failed" ||
            initState === "none")) ||
          (wsPrevInit === null && initState === "ready")) {
        wsUI.refresh();
      }
      wsPrevInit = initState;
    } catch (e) { /* Agent kurz down — weiter pollen */ }
    if ((!assetsPending && !building) || wsPollTries >= 120) {
      wsStopPolling();
      const syncBtn = document.getElementById("ws-sync-btn");
      if (syncBtn) { syncBtn.disabled = false; syncBtn.textContent = "🔄 Sync & Init jetzt ausführen"; }
    }
  }
  function wsStopPolling() {
    if (wsPollTimer) { clearInterval(wsPollTimer); wsPollTimer = null; }
  }
  function wsStartPolling() {
    if (!WS_TASK_ID) return;
    if (wsPollTimer) { wsPollTick(); return; }
    wsPollTries = 0;
    wsPollTimer = setInterval(wsPollTick, 3000);
    wsPollTick();
  }

  // Init-Status am Seitenanfang: Status-Chips rendern; bei pending/building pollen.
  wsRenderAssetStatus(WS_ASSETS_INIT);
  if (WS_ASSETS_INIT && /"pending"|"building"/.test(WS_ASSETS_INIT)) {
    wsStartPolling();
  }

  document.getElementById("ws-sync-btn").addEventListener("click", async () => {
    if (!WS_TASK_ID) {
      showToast("Erst die Aufgabe erstellen (speichern), dann Sync/Init ausführen.", "warning");
      return;
    }
    const btn = document.getElementById("ws-sync-btn");
    btn.disabled = true;
    btn.textContent = "⏳ Sync & Init läuft …";

      const res = await fetch(WS_TUTOR_FILES_API + "/sync", { method: "POST", credentials: "same-origin" });
      const data = await res.json();
      if (!res.ok) throw new Error(data.detail || res.status);
      wsStartPolling();
    } catch (err) {
      showToast("Sync fehlgeschlagen: " + err.message, "error");
      btn.disabled = false;
      btn.textContent = "🔄 Sync & Init jetzt ausführen";
    }
  });

  document.getElementById("ws-init-stop-btn").addEventListener("click", async () => {
    if (!confirm("Init-Build abbrechen? (Halb-fertige Init-Daten werden bereinigt.)")) return;
    try {
      const res = await fetch(WS_TUTOR_FILES_API + "/init/stop", { method: "POST", credentials: "same-origin" });
      const data = await res.json();
      if (!res.ok) throw new Error(data.detail || res.status);
      showToast("Abbruch angefordert — der Status aktualisiert sich in wenigen Sekunden.", "success");
      wsStartPolling();
    } catch (err) {
      showToast("Abbruch fehlgeschlagen: " + err.message, "error");
    }
  });

  // ─── Musterlösung-Testlauf (ephemer, Live-Log) ────────────────
  const wsTestRun = { active: null };

  function wsTestRunSetEnabled(enabled) {
    const btn = document.getElementById("ws-test-solution-btn");
    if (btn) btn.disabled = !enabled;
  }

  async function wsTestRunStart() {
    if (wsTestRun.active) return;
    if (!WS_TASK_ID) {
      showToast("Erst die Aufgabe erstellen (speichern), dann die Musterlösung testen.", "warning");
      return;
    }
    const stopBtn = document.getElementById("ws-test-run-stop-btn");
    const terminal = document.getElementById("ws-test-run-terminal");
    const termText = document.getElementById("ws-test-run-terminal-text");
    wsTestRunSetEnabled(false);
    stopBtn.classList.add("hidden");
    terminal.classList.remove("hidden");
    termText.textContent = "Testlauf wird gestartet (frische ephemere Umgebung) …";
    termText.style.color = "#9ca3af";
    try {
      const res = await fetch(WS_TUTOR_FILES_API + "/test-run", {
        method: "POST", credentials: "same-origin",
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data.detail || res.status);
      wsTestRun.active = { runId: data.run_id, startedAt: Date.now() };
      stopBtn.classList.remove("hidden");
      wsTestRunPoll();
    } catch (err) {
      termText.textContent = "Fehler: " + err.message;
      termText.style.color = "#f87171";
      wsTestRunSetEnabled(true);
    }
  }

  function wsTestRunPoll() {
    const active = wsTestRun.active;
    if (!active) return;
    const runId = active.runId;
    (async () => {
      try {
        const res = await fetch(WS_TUTOR_FILES_API + "/test-runs/" + runId, { credentials: "same-origin" });
        const st = await res.json();
        if (!res.ok) throw new Error(st.detail || res.status);
        const norm = (v) => Array.isArray(v) ? v.join("\n") : String(v || "");
        const out = norm(st.stdout);
        const err = norm(st.stderr);
        const combined = [out, err].filter(Boolean).join("\n");
        const termText = document.getElementById("ws-test-run-terminal-text");
        if (st.status === "running" || st.status === "queued") {
          const s = Math.floor((Date.now() - active.startedAt) / 1000);
          const mm = String(Math.floor(s / 60)).padStart(2, "0");
          const ss = String(s % 60).padStart(2, "0");
          document.getElementById("ws-test-run-duration").textContent =
            (st.status === "queued") ? "⏳ wartet auf Start (GPU-Queue) …" : "⏳ läuft seit " + mm + ":" + ss;
          termText.textContent = combined || "⏳ Läuft … (noch keine Ausgabe)";
          termText.style.color = combined ? "#86efac" : "#9ca3af";
          if (combined) termText.scrollTop = termText.scrollHeight;
          wsTestRun.active.timer = setTimeout(wsTestRunPoll, 2500);
        } else {
          wsTestRunFinish({
            status: st.status,
            exit_code: st.exit_code,
            stdout: st.stdout,
            stderr: st.stderr,
            timed_out: st.status === "timeout",
          });
        }
      } catch (err) {
        // Agent kurz down? Weiter pollen (Backend fällt auf DB zurück).
        wsTestRun.active.timer = setTimeout(wsTestRunPoll, 5000);
      }
    })();
  }

  function wsTestRunFinish(data) {
    const active = wsTestRun.active;
    wsTestRun.active = null;
    if (active && active.timer) clearTimeout(active.timer);
    document.getElementById("ws-test-run-stop-btn").classList.add("hidden");
    wsTestRunSetEnabled(true);
    const termText = document.getElementById("ws-test-run-terminal-text");
    const durationEl = document.getElementById("ws-test-run-duration");
    let output = "";
    if (data.stdout && String(data.stdout).trim()) output += String(data.stdout);
    if (data.stderr && String(data.stderr).trim()) {
      if (output) output += "\n\n";
      output += "--- FEHLER ---\n" + data.stderr.trim();
    }
    if (!output.trim()) output = "(Keine Ausgabe)";
    if (data.timed_out || data.status === "timeout") {
      output += "\n\n[⏰ TIMEOUT: Zeitlimit erreicht]";
    }
    termText.textContent = output;
    const failed = (data.exit_code != null && data.exit_code !== 0) || data.timed_out;
    termText.style.color = failed ? "#f87171" : "#86efac";
    termText.scrollTop = termText.scrollHeight;
    const s = active ? Math.floor((Date.now() - active.startedAt) / 1000) : 0;
    durationEl.textContent = "⏱ " + s + " s · Exit-Code: " + (data.exit_code != null ? data.exit_code : "—");
  }

  // (task_detail_base.html ruft window._workspaceAiApply(data) auf,

  window._workspaceAiApply = function (data) {
    const applied = [];
    // Umgebungsfelder (flache Keys aus der LLM-Antwort)
    const envKeys = ["workspace_timeout", "workspace_cpu", "workspace_memory",
                     "workspace_disk_quota", "workspace_internet",
                     "workspace_main_file"];
    if (envKeys.some(k => data[k] !== undefined)) {
      if (data.workspace_timeout != null) wsTimeoutInput.value = data.workspace_timeout;
      if (data.workspace_cpu != null) wsCpuInput.value = data.workspace_cpu;
      if (data.workspace_memory != null) wsMemoryInput.value = wsMemToGb(data.workspace_memory);
      if (data.workspace_disk_quota != null) wsDiskInput.value = data.workspace_disk_quota;
      if (data.workspace_internet !== undefined) wsInternetChk.checked = !!data.workspace_internet;
      if (data.workspace_main_file !== undefined) wsUI.setMainFile(data.workspace_main_file || "");
      applied.push("Umgebung");
    }
    // Engine + Image-Spec (LLM wählt aus registrierten Engines/Specs)
    // Erst die Spec setzen (triggert das asynchrone Re-Filtern der Engine-Optionen),
    // dann die Engine — sonst würde die Engine gegen den alten Filter geprüft.
    let specChanged = false;
    if (data.workspace_image !== undefined) {
      const newImage = data.workspace_image || "";
      if (newImage && Array.from(wsSpecSel.options).some(o => o.value === newImage)) {
        wsSpecSel.value = newImage;
        applied.push("Image-Spec");
        specChanged = true;
      } else if (!newImage) {
        wsSpecSel.value = "";
        specChanged = true;
      }
      if (specChanged) {
        wsUpdateSpecInfo();
        wsRefreshInstalled();
      }
    }
    if (data.workspace_engines !== undefined) {
      const eng = (data.workspace_engines || [])[0];
      const tryApplyEngine = () => {
        if (eng && Array.from(wsEngineSel.options).some(o => o.value === eng)) {
          wsEngineSel.value = eng;
          return true;
        }
        return false;
      };
      if (tryApplyEngine()) {
        applied.push("Engine");
      } else if (eng && specChanged) {
        // Engine-Optionen werden gerade neu gefiltert → nach Abschluss erneut versuchen
        wsEngineRefreshP.then(tryApplyEngine);
      }
    }
    // Neuartige Image-Spec vorgeschlagen (keine passende vorhanden) — Specs
    // werden in den Einstellungen (Kurs/Admin → Compute/Workspace) angelegt.
    if (data.proposed_image_spec && data.proposed_image_spec.dockerfile) {
      showToast("Das LLM hat eine neue Image-Spec („" + (data.proposed_image_spec.name || "ohne Name") +
        "“) vorgeschlagen — bitte in den Einstellungen → Compute/Workspace anlegen.", "warning");
    }
    const nWritten = (data.workspace_files || []).length;
    const nEdited = (data.workspace_files_edited || []).length;
    const nDeleted = (data.workspace_files_deleted || []).length;
    if (nWritten || nEdited || nDeleted) {
      wsUI.refresh();
      const parts = [];
      if (nWritten) parts.push(nWritten + " neu/geschrieben");
      if (nEdited) parts.push(nEdited + " lokal editiert");
      if (nDeleted) parts.push(nDeleted + " gelöscht");
      applied.push("Dateien: " + parts.join(", "));
    }
    (data.workspace_file_warnings || []).forEach((w) => {
      showToast("⚠️ " + w, "warning");
    });
    return applied;
  };
